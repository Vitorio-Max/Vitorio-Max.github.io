-- 1. Establecimientos (Mercadona, Carrefour, Frutería...)
CREATE TABLE establecimientos (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  nombre TEXT NOT NULL UNIQUE,
  color_hex TEXT DEFAULT '#3B82F6', -- Para identificar visualmente cada tienda
  created_at TIMESTAMPTZ DEFAULT now()
);

-- 2. Ubicaciones por Establecimiento (Define la ruta física)
CREATE TABLE ubicaciones (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  establecimiento_id UUID REFERENCES establecimientos(id) ON DELETE CASCADE,
  nombre TEXT NOT NULL, -- Ej: "Fruta y Verdura", "Pasillo 3 (Lácteos)", "Congelados"
  orden_recorrido INT NOT NULL DEFAULT 10, -- CLAVE: 10 = Entrada, 20 = Pasillo 1... 90 = Cajas
  UNIQUE(establecimiento_id, nombre)
);

-- 3. Catálogo de Productos
CREATE TABLE productos (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  nombre TEXT NOT NULL,
  precio DECIMAL(10,2) DEFAULT 0.00,
  establecimiento_id UUID REFERENCES establecimientos(id) ON DELETE SET NULL,
  ubicacion_id UUID REFERENCES ubicaciones(id) ON DELETE SET NULL,
  es_basico BOOLEAN DEFAULT false, -- Pestaña de favoritos/básicos
  unidad_medida TEXT DEFAULT 'ud', -- Propuesta extra: 'ud', 'kg', 'pack', 'L'
  notas_habituales TEXT, -- Propuesta extra: Ej. "Sin lactosa", "Marca Hacendado"
  created_at TIMESTAMPTZ DEFAULT now()
);

-- 4. Lista de la Compra Activa
CREATE TABLE lista_compra (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  producto_id UUID REFERENCES productos(id) ON DELETE CASCADE,
  cantidad DECIMAL(10,2) DEFAULT 1,
  comprado BOOLEAN DEFAULT false,
  nota_puntual TEXT, -- Ej: "Que no estén muy maduros"
  added_at TIMESTAMPTZ DEFAULT now(),
  UNIQUE(producto_id) -- Evita duplicados; si lo añades de nuevo, sumas cantidad
);
