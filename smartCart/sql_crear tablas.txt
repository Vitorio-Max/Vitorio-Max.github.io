-- Habilitar extensión para búsquedas insensibles a tildes/mayúsculas si se desea
CREATE EXTENSION IF NOT EXISTS pg_trgm;

-- 1. Tabla de Establecimientos
CREATE TABLE establecimientos (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  nombre TEXT NOT NULL UNIQUE,
  color_hex TEXT NOT NULL DEFAULT '#10B981',
  created_at TIMESTAMPTZ DEFAULT now()
);

-- 2. Tabla de Ubicaciones (Pasillos / Zonas ordenadas)
CREATE TABLE ubicaciones (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  establecimiento_id UUID NOT NULL REFERENCES establecimientos(id) ON DELETE CASCADE,
  nombre TEXT NOT NULL,
  orden_recorrido INTEGER NOT NULL DEFAULT 1,
  created_at TIMESTAMPTZ DEFAULT now(),
  UNIQUE(establecimiento_id, nombre)
);

CREATE INDEX idx_ubicaciones_recorrido ON ubicaciones(establecimiento_id, orden_recorrido ASC);

-- 3. Tabla de Productos (Catálogo + Inventario de Despensa)
CREATE TABLE productos (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  nombre TEXT NOT NULL UNIQUE,
  precio NUMERIC(10, 2) NOT NULL DEFAULT 0.00 CHECK (precio >= 0),
  unidad_medida TEXT NOT NULL DEFAULT 'ud' CHECK (unidad_medida IN ('ud', 'kg', 'pack', 'l')),
  categoria_icono TEXT NOT NULL DEFAULT '🛒',
  establecimiento_id UUID REFERENCES establecimientos(id) ON DELETE SET NULL,
  ubicacion_id UUID REFERENCES ubicaciones(id) ON DELETE SET NULL,
  es_basico BOOLEAN NOT NULL DEFAULT false,
  stock_actual INTEGER NOT NULL DEFAULT 0 CHECK (stock_actual >= 0),
  stock_minimo INTEGER NOT NULL DEFAULT 1 CHECK (stock_minimo >= 0),
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now()
);

CREATE INDEX idx_productos_nombre ON productos USING gin (nombre gin_trgm_ops);
CREATE INDEX idx_productos_basicos ON productos(es_basico) WHERE es_basico = true;

-- 4. Tabla de Lista de Compra Activa
CREATE TABLE lista_compra (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  producto_id UUID NOT NULL UNIQUE REFERENCES productos(id) ON DELETE CASCADE,
  cantidad INTEGER NOT NULL DEFAULT 1 CHECK (cantidad > 0),
  comprado BOOLEAN NOT NULL DEFAULT false,
  notas TEXT,
  created_at TIMESTAMPTZ DEFAULT now()
);

CREATE INDEX idx_lista_comprado ON lista_compra(comprado);

-- Políticas RLS (Acceso completo para uso personal/familiar mediante anon/authenticated key)
ALTER TABLE establecimientos ENABLE ROW LEVEL SECURITY;
ALTER TABLE ubicaciones ENABLE ROW LEVEL SECURITY;
ALTER TABLE productos ENABLE ROW LEVEL SECURITY;
ALTER TABLE lista_compra ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Acceso total establecimientos" ON establecimientos FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Acceso total ubicaciones" ON ubicaciones FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Acceso total productos" ON productos FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Acceso total lista_compra" ON lista_compra FOR ALL USING (true) WITH CHECK (true);

-- Función auxiliar para reordenar pasillos al insertar uno intermedio
CREATE OR REPLACE FUNCTION desplazar_orden_ubicaciones(p_establecimiento_id UUID, p_a_partir_de INT)
RETURNS VOID AS $$
BEGIN
  UPDATE ubicaciones
  SET orden_recorrido = orden_recorrido + 1
  WHERE establecimiento_id = p_establecimiento_id
    AND orden_recorrido >= p_a_partir_de;
END;
$$ LANGUAGE plpgsql;
